cmake_minimum_required(VERSION 3.20)
project(qp_scale_replay LANGUAGES CXX)
find_package(Python 3.12 REQUIRED COMPONENTS Interpreter Development.Module Development.SABIModule)
find_package(nanobind CONFIG REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(trajopt_ifopt CONFIG REQUIRED)
find_package(piqp CONFIG REQUIRED)
set(SQP_SOURCE_INCLUDE "" CACHE PATH "Corrected SQP source include")
set(SQP_LIBRARY "" CACHE FILEPATH "Corrected SQP dynamic library")
if(NOT SQP_SOURCE_INCLUDE OR NOT SQP_LIBRARY)
  message(FATAL_ERROR "Provide SQP_SOURCE_INCLUDE and SQP_LIBRARY")
endif()
nanobind_add_module(_qp_scale_replay STABLE_ABI qp_scale_replay.cpp)
target_compile_features(_qp_scale_replay PRIVATE cxx_std_17)
target_include_directories(_qp_scale_replay BEFORE PRIVATE "${SQP_SOURCE_INCLUDE}")
target_link_libraries(_qp_scale_replay PRIVATE "${SQP_LIBRARY}" trajopt::trajopt_ifopt piqp::piqp Eigen3::Eigen)
