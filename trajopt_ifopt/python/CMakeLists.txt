cmake_minimum_required(VERSION 3.20)
project(quintic_spline_binding LANGUAGES CXX)
find_package(Python 3.12 REQUIRED COMPONENTS Interpreter Development.Module Development.SABIModule)
find_package(nanobind CONFIG REQUIRED)

find_package(trajopt_ifopt CONFIG REQUIRED)
nanobind_add_module(_quintic_spline STABLE_ABI quintic_spline_bindings.cpp)
target_compile_features(_quintic_spline PRIVATE cxx_std_17)
target_link_libraries(_quintic_spline PRIVATE trajopt::trajopt_ifopt Eigen3::Eigen)
